\chapter*{Notation and Abbreviations}
\addcontentsline{toc}{chapter}{Notation and Abbreviations}
\begin{tabularx}{\textwidth}{@{}l X@{}}
\toprule
\multicolumn{2}{@{}l}{\textbf{Network and losses}}\\
$x\in\R^{784}$, $y\in\{0,\dots,9\}$ & input image and class label\\
$\Flow$, $\Fup$ & lower and upper part of the network; $h=\Flow(x)\in\R^{128}$ is the middle state\\
$z=\Fup(h)$, $p=\mathrm{softmax}(z)$ & logits and class probabilities\\
$\onehot$ & one-hot vector of the label $y$\\
$G$ & learned backward (inverse) network that maps the output space to the middle state\\
$b = h + G(\onehot)-G(p)$ & difference-corrected target for $h$\\
$\Ltask,\ \Linv,\ \Lmeet$ & cross-entropy loss, inverse-network loss, meeting loss\\
$\lambda_{\mathrm{meet}},\ \lambda_{\mathrm{inv}}$ & weights of the two auxiliary terms\\
\addlinespace
\multicolumn{2}{@{}l}{\textbf{Selective backward pass}}\\
$B$ & batch size ($B=128$ unless stated)\\
$\phi$, $k=\lceil\phi B\rceil$ & kept fraction and number of kept rows\\
S30, S20, S20w0, \dots & selective-backward arms: kept fraction in percent, then \code{w}$n$ = $n$ warm-up steps with the full backward pass (default 100)\\
\addlinespace
\multicolumn{2}{@{}l}{\textbf{Evaluation}}\\
FLOPs-to-target & cumulative counted FLOPs at the first of two consecutive evaluations at or above the target validation accuracy\\
$\GFLOPs$ & $10^9$ floating-point operations (a multiply--add counts as 2)\\
CI & 95\% (or stated) paired-bootstrap interval\\
\addlinespace
\multicolumn{2}{@{}l}{\textbf{Abbreviations}}\\
BP, DTP, SDTP & backpropagation; difference target propagation; simple difference target propagation\\
DSN, OHEM & deeply supervised nets; online hard example mining\\
SB, SBT & Selective-Backprop; its tuned configuration\\
MLP, FLOPs & multilayer perceptron; floating-point operations\\
\bottomrule
\end{tabularx}
